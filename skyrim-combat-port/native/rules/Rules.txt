# TGCombatPort — правила механики.
# Файл ставится в Data\SKSE\Plugins\TGCombatPort\Rules.txt.
# Плагин читает его при запуске игры и печатает в лог: "Rules.txt [...]: правил N".
#
# Каждая секция [имя] — одно правило. «Хочу, чтобы такая-то атака такого-то врага
# делала то же самое» — это новая секция здесь, код менять не нужно.
#
# ── условия ────────────────────────────────────────────────────────────────────
#   trigger         = hit, bash      hit — настоящее попадание; bash — наш заказанный баш
#   hit-flags       = blocked, power, bash, sneak   (все перечисленные должны быть)
#   hit-not-flags   = power          (ни одного из перечисленных быть не должно)
#   attack.kind     = melee, ranged, spell, shout, unarmed, any
#   attacker.race     = <раса>        например BearRace; список через запятую
#   attacker.weapon   = fists, sword, dagger, axe, mace, greatsword, battleaxe, warhammer,
#                       bow, staff, crossbow | any
#                       Двуручный топор и двуручный молот у движка ОДИН тип (kTwoHandAxe),
#                       поэтому они различаются кейвордами: battleaxe = WeapTypeBattleaxe,
#                       warhammer = WeapTypeWarhammer (так же их различают ванильные перки).
#   attacker.shield   = yes | no | any
#   hit.source      = <форма>        то, чем бьют: оружие, заклинание, крик (TESHitEvent::source)
#   hit.projectile  = <форма>        снаряд: болт, стрела, файрбол (TESHitEvent::projectile)
#                       «any»/«none» (есть снаряд / магия без снаряда) пока НЕ поддерживаются —
#                       это следующая правка кода, см. задачу «магия: снаряд против AoE».
#   player.blocking = yes | no | any
#   player.shield   = yes | no | any
#   player.ward     = yes | no | any     поднят ли у игрока вард (Малый Оберег и др.)
#   reach           = yes | no       проверять дистанцию fCombatBashReach
#   cooldown        = секунды между срабатываниями правила
#   chip            = сколько здоровья снимает эффект chip (по умолчанию 10)
#   do              = unblock, stagger, bash   (порядок сохраняется)
#
# ── как записывать формы (раса, оружие, снаряд, заклинание) ────────────────────
#   BearRace                — EditorID из Creation Kit (самый удобный способ)
#   Skyrim.esm|0x131E9      — плагин и локальный FormID (не зависит от порядка загрузки)
#   0x000131E9              — полный FormID как он есть в этой игре
#   Значения для конкретного удара берутся из строки лога "hit: … race=… source=… projectile=…".
#
# ── эффекты ────────────────────────────────────────────────────────────────────
#   unblock  — снять блок у игрока и удерживать снятым ~1 с
#   stagger  — стаггер спеллом от имени бьющего (дезориентация)
#   bash     — заказать бьющему баш
#   chip     — снять N здоровья СКВОЗЬ блок (блок остаётся, оглушения нет)
#
# ── что решено 22.11.2026 (правки пользователя) ────────────────────────────────
#   * одноручный топор НЕ оглушает и блок оружием НЕ пробивает;
#   * двуручный топор НЕ оглушает, блок не пробивает, но может нанести травму
#     (эффект wound — в коде ещё нет);
#   * двуручный меч на зажатие — широкий выпад по площади, большой урон без эффектов,
#     поэтому блок оружием он НЕ пробивает;
#   * обычное двуручное у противников не оглушает и блок не пробивает;
#   * щит дезориентирует только того, кто НЕ блокирует; сквозь блок оружием щит
#     не пробивает и блок не снимает;
#   * лук и арбалет простреливают блок оружием насквозь, щит их держит;
#   * вся магия проходит сквозь блок оружием; снаряд гасится щитом, AoE (столб,
#     поток, разряд без снаряда) пробивает и щит.

[settings]
# effects = off — правила считаются и пишутся в лог, но движок не трогаем вообще
# (в логе появляется «would fire» вместо «rule fired»). Это способ проверить,
# виноват ли эффект плагина: если игра падает и в этом режиме, дело не в эффектах.
effects = on

[shield-bash-staggers-unblocked]
# Щит дезориентирует только того, у кого нет блока вообще.
# Если игрок блокирует оружием — щит не пробивает и блок не снимает: правила на этот
# случай СОЗНАТЕЛЬНО НЕТ.
trigger = hit, bash
player.blocking = no
do = stagger
cooldown = 3

[onehand-mace-power-breaks-shield-block]
# Щит игрока пробивает силовая атака булавы.
trigger = hit
hit-flags = blocked, power
attacker.weapon = mace
player.blocking = yes
player.shield = yes
do = unblock, stagger
cooldown = 4

[twohand-warhammer-breaks-shield-block]
# Двуручный МОЛОТ пробивает щит. Двуручный топор — нет: у него своё правило ниже,
# а различаются они кейвордом WeapTypeWarhammer.
trigger = hit
hit-flags = blocked, power
attacker.weapon = warhammer
player.blocking = yes
player.shield = yes
do = unblock, stagger
cooldown = 4

# ── уколы: силовая атака кинжала и меча ────────────────────────────────────────

[dagger-thrust-through-any-block]
# Кинжал: укол проходит и блок оружием, и блок щитом (урон, без оглушения).
trigger = hit
hit-flags = blocked, power
attacker.weapon = dagger
player.blocking = yes
do = chip
chip = 10
cooldown = 0

[sword-thrust-through-weapon-block]
# Меч: укол проходит блок оружием, но щит его останавливает.
trigger = hit
hit-flags = blocked, power
attacker.weapon = sword
player.blocking = yes
player.shield = no
do = chip
chip = 8
cooldown = 0

[dagger-thrust-through-ward]
# Кинжал проходит и сквозь оберег: магия останавливает заклинания, а не тонкий клинок.
trigger = hit
hit-flags = blocked, power
attacker.weapon = dagger
player.ward = yes
do = chip
chip = 6
cooldown = 0

# ── стрельба ─────────────────────────────────────────────────────────────────

[arrow-and-bolt-through-weapon-block]
# Лук и арбалет простреливают блок оружием насквозь. Щит держит: player.shield = no.
trigger = hit
hit-flags = blocked
attack.kind = ranged
attacker.weapon = bow, crossbow
player.blocking = yes
player.shield = no
do = chip
chip = 8
cooldown = 0

# ── магия ────────────────────────────────────────────────────────────────────

[spell-through-weapon-block]
# Вся магия проходит сквозь блок оружием; вард её гасит, поэтому player.ward = no.
trigger = hit
hit-flags = blocked
attack.kind = spell
player.blocking = yes
player.shield = no
player.ward = no
do = chip
chip = 6
cooldown = 0

# ── выключено: ждёт правки кода ───────────────────────────────────────────────
#
# 1) «Снаряд гасится щитом, AoE пробивает щит».
#    Нужны значения hit.projectile = any (есть снаряд) и none (магия без снаряда).
#    После этого включаются два правила:
#
#    [spell-projectile-stopped-by-shield]     # снаряд: щит держит → ничего не делаем
#    [spell-aoe-breaks-shield]                # AoE: пробивает щит
#    trigger = hit
#    hit-flags = blocked
#    attack.kind = spell
#    player.blocking = yes
#    player.shield = yes
#    hit.projectile = none
#    do = chip
#    chip = 8
#    cooldown = 0
#
# 2) «Двуручный топор наносит травму, но не оглушает».
#    Нужен эффект wound (ранения по частям тела), его пока нет:
#
#    [twohand-axe-wounds]
#    trigger = hit
#    hit-flags = blocked, power
#    attacker.weapon = battleaxe
#    do = wound
#    cooldown = 8
